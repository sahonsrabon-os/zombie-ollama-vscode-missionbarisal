import os
import json
import requests

# ============================================

# ============================================
SOCK = '/tmp/ollama.sock'
session = requests_unixsocket.Session()
encoded = SOCK.replace('/', '%2F')
base = f'http+unix://{encoded}'

# Status
r = session.get(f'{base}/api/status')
print(f"Status: {r.json()['status']}")

# Models
r = session.get(f'{base}/api/tags')
models = [m['name'] for m in r.json()['models']]
print(f"Models: {models}")

# Generate
r = session.post(f'{base}/api/generate', json={
    'model': 'deepseek-r1:1.5b',
    'prompt': 'What is AI?',
    'stream': False
})
print(f"Response: {r.json()['response']}")

# Chat
r = session.post(f'{base}/api/chat', json={
    'model': 'deepseek-r1:1.5b',
    'messages': [{'role': 'user', 'content': 'Hello!'}],
    'stream': False
})
print(f"Chat: {r.json()['message']['content']}")

# OpenAI Completions
r = session.post(f'{base}/v1/completions', json={
    'model': 'deepseek-r1:1.5b',
    'prompt': 'def fibonacci(',
    'max_tokens': 50
})
print(f"Completions: {r.json()['choices'][0]['text']}")

# OpenAI Chat Completions
r = session.post(f'{base}/v1/chat/completions', json={
    'model': 'deepseek-r1:1.5b',
    'messages': [{'role': 'user', 'content': 'What is Python?'}],
    'max_tokens': 50
})
print(f"Chat Completions: {r.json()['choices'][0]['message']['content']}")

# Embeddings
r = session.post(f'{base}/v1/embeddings', json={
    'model': 'deepseek-r1:1.5b',
    'input': 'Hello world'
})
dims = len(r.json()['data'][0]['embedding'])
print(f"Embedding dims: {dims}")

# System
r = session.get(f'{base}/api/system')
info = r.json()
print(f"CPU: {info['cpu_count']} | Memory: {info['mem_usage']}")

VS Code Extension (src/extension.ts)

TypeScript Extension

import * as vscode from 'vscode';
import axios from 'axios';

// ============================================
// Configuration
// ============================================
const HTTP_BASE = 'http://localhost:5000';
const UDS_PATH = '/tmp/ollama.sock';
const DEFAULT_MODEL = 'deepseek-r1:1.5b';

// Use UDS if available (faster), fallback to TCP
let useUDS = false;
let baseUrl = HTTP_BASE;

// For UDS: use requests_unixsocket equivalent in Node.js
// In VS Code extension, prefer TCP for simplicity
// For UDS, use: socat UNIX-CONNECT:$UDS_PATH TCP:localhost:80

// ============================================
// API Functions
// ============================================
async function getStatus() {
  const res = await axios.get(`${baseUrl}/api/status`);
  return res.data;
}

async function getModels() {
  const res = await axios.get(`${baseUrl}/api/tags`);
  return res.data.models.map((m: any) => m.name);
}

async function getRunningModels() {
  const res = await axios.get(`${baseUrl}/api/ps`);
  return res.data.models || [];
}

async function generate(model: string, prompt: string, stream = false) {
  const res = await axios.post(`${baseUrl}/api/generate`, {
    model, prompt, stream
  });
  return res.data.response;
}

async function chat(model: string, messages: any[], stream = false) {
  const res = await axios.post(`${baseUrl}/api/chat`, {
    model, messages, stream
  });
  return res.data.message.content;
}

async function openaiChatCompletion(model: string, messages: any[], maxTokens = 256) {
  const res = await axios.post(`${baseUrl}/v1/chat/completions`, {
    model, messages, max_tokens: maxTokens
  });
  return res.data.choices[0].message.content;
}

async function openaiCompletion(model: string, prompt: string, maxTokens = 50) {
  const res = await axios.post(`${baseUrl}/v1/completions`, {
    model, prompt, max_tokens: maxTokens
  });
  return res.data.choices[0].text;
}

async function getSystemInfo() {
  const res = await axios.get(`${baseUrl}/api/system`);
  return res.data;
}

// ============================================
// VS Code Commands
// ============================================
export function activate(context: vscode.ExtensionContext) {
  
  // Command: Check Status
  context.subscriptions.push(
    vscode.commands.registerCommand('ollama.status', async () => {
      try {
        const status = await getStatus();
        vscode.window.showInformationMessage(
          `Status: ${status.status} | Model: ${status.model} | Ollama: ${status.ollama_status}`
        );
      } catch (e: any) {
        vscode.window.showErrorMessage(`Error: ${e.message}`);
      }
    })
  );

  // Command: List Models
  context.subscriptions.push(
    vscode.commands.registerCommand('ollama.listModels', async () => {
      try {
        const models = await getModels();
        const selected = await vscode.window.showQuickPick(models, {
          placeHolder: 'Select a model'
        });
        if (selected) {
          vscode.window.showInformationMessage(`Selected: ${selected}`);
        }
      } catch (e: any) {
        vscode.window.showErrorMessage(`Error: ${e.message}`);
      }
    })
  );

  // Command: Generate Text (replace selection)
  context.subscriptions.push(
    vscode.commands.registerCommand('ollama.generate', async () => {
      const editor = vscode.window.activeTextEditor;
      if (!editor) {
        vscode.window.showWarningMessage('No active editor');
        return;
      }

      const selection = editor.selection;
      const selectedText = editor.document.getText(selection);
      const prompt = selectedText || await vscode.window.showInputBox({
        prompt: 'Enter your prompt'
      });

      if (!prompt) return;

      await vscode.window.withProgress({
        location: vscode.ProgressLocation.Notification,
        title: 'Generating...',
        cancellable: false
      }, async () => {
        try {
          const result = await generate(DEFAULT_MODEL, prompt);
          await editor.edit(editBuilder => {
            editBuilder.replace(selection, result);
          });
        } catch (e: any) {
          vscode.window.showErrorMessage(`Error: ${e.message}`);
        }
      });
    })
  );

  // Command: Chat (new panel)
  context.subscriptions.push(
    vscode.commands.registerCommand('ollama.chat', async () => {
      const input = await vscode.window.showInputBox({
        prompt: 'Enter your message'
      });
      if (!input) return;

      await vscode.window.withProgress({
        location: vscode.ProgressLocation.Notification,
        title: 'Thinking...',
        cancellable: false
      }, async () => {
        try {
          const response = await chat(DEFAULT_MODEL, [
            { role: 'user', content: input }
          ]);
          vscode.window.showInformationMessage(response);
        } catch (e: any) {
          vscode.window.showErrorMessage(`Error: ${e.message}`);
        }
      });
    })
  );

  // Command: Open Admin Panel
  context.subscriptions.push(
    vscode.commands.registerCommand('ollama.admin', () => {
      vscode.env.openExternal(vscode.Uri.parse(`${baseUrl}/admin`));
    })
  );
}

export function deactivate() {}

Known Issues

System Logs Not Generated

Docker container logs থেকে সিস্টেম লগ সঠিকভাবে জেনারেট হচ্ছে না। এটি একটি পরিচিত সমস্যা যেটি এখনও সমাধান হয়নি।

Workaround: Flask app এর ভিতরে logging যোগ করতে হবে। 

# app.py তে logging যোগ করুন:
import logging
from logging.handlers import RotatingFileHandler

# File handler
handler = RotatingFileHandler('/app/logs/app.log', maxBytes=10*1024*1024, backupCount=5)
handler.setFormatter(logging.Formatter('%(asctime)s [%(levelname)s] %(message)s'))
app.logger.addHandler(handler)
app.logger.setLevel(logging.INFO)

# এরপর সব জায়গায় app.logger.info() ব্যবহার করুন
app.logger.info(f"Request: {request.method} {request.path}")
